import { router } from "expo-router";
import { useEffect, useMemo, useState } from "react";
import {
  ActivityIndicator,
  KeyboardAvoidingView,
  Platform,
  Pressable,
  ScrollView,
  StyleSheet,
  Text,
  TextInput,
  useWindowDimensions,
  View,
} from "react-native";
import { SafeAreaView } from "react-native-safe-area-context";

import { supabase } from "../../lib/supabase";

type ItemOrcamento = {
  id: string;
  descricao: string;
  quantidade: string;
  valorUnitario: string;
  servicoId: string | null;
};

type ClienteCadastrado = {
  id: string;
  nome: string;
  telefone: string | null;
  email: string | null;
  endereco: string | null;
  cidade: string | null;
  estado: string | null;
};

type ObraCadastrada = {
  id: string;
  nome: string;
  cliente_id: string | null;
  status: string;
};

type ServicoCadastrado = {
  id: string;
  nome: string;
  unidade: string;
  valor_unitario: number | string;
  ativo: boolean;
};

function criarIdLocal() {
  return `${Date.now()}-${Math.random().toString(36).slice(2, 9)}`;
}

function normalizarNumero(valor: string) {
  const limpo = valor
    .trim()
    .replace(/\s/g, "")
    .replace(/\./g, "")
    .replace(",", ".");

  const numero = Number(limpo);

  if (!Number.isFinite(numero)) {
    return 0;
  }

  return numero;
}

function formatarMoeda(valor: number) {
  return new Intl.NumberFormat("pt-BR", {
    style: "currency",
    currency: "BRL",
  }).format(valor);
}

export default function NovoOrcamentoScreen() {
  const { width } = useWindowDimensions();
  const isDesktop = width >= 900;

  const [clientesCadastrados, setClientesCadastrados] = useState<
    ClienteCadastrado[]
  >([]);
  const [obrasCadastradas, setObrasCadastradas] = useState<ObraCadastrada[]>([]);
  const [servicosCadastrados, setServicosCadastrados] = useState<
    ServicoCadastrado[]
  >([]);
  const [clienteSelecionadoId, setClienteSelecionadoId] = useState<
    string | null
  >(null);
  const [obraSelecionadaId, setObraSelecionadaId] = useState<string | null>(
    null
  );
  const [carregandoCadastros, setCarregandoCadastros] = useState(true);
  const [validandoSessao, setValidandoSessao] = useState(true);
  const [redirecionandoLogin, setRedirecionandoLogin] = useState(false);

  const [clienteNome, setClienteNome] = useState("");
  const [clienteTelefone, setClienteTelefone] = useState("");
  const [clienteCep, setClienteCep] = useState("");
  const [clienteLogradouro, setClienteLogradouro] = useState("");
  const [clienteNumero, setClienteNumero] = useState("");
  const [clienteComplemento, setClienteComplemento] = useState("");
  const [clienteBairro, setClienteBairro] = useState("");
  const [clienteCidade, setClienteCidade] = useState("");
  const [clienteUf, setClienteUf] = useState("");
  const [clienteEstado, setClienteEstado] = useState("");
  const [buscandoCep, setBuscandoCep] = useState(false);
  const [erroCep, setErroCep] = useState("");
  const [clienteEmail, setClienteEmail] = useState("");

  const [titulo, setTitulo] = useState("");
  const [descricao, setDescricao] = useState("");
  const [observacoes, setObservacoes] = useState("");
  const [validadeDias, setValidadeDias] = useState("15");
  const [desconto, setDesconto] = useState("0");
  const [acrescimo, setAcrescimo] = useState("0");
  const [formaPagamento, setFormaPagamento] = useState("");
  const [condicoesPagamento, setCondicoesPagamento] = useState("");

  const [itens, setItens] = useState<ItemOrcamento[]>([
    {
      id: criarIdLocal(),
      descricao: "",
      quantidade: "1",
      valorUnitario: "",
      servicoId: null,
    },
  ]);

  const [salvando, setSalvando] = useState(false);
  const [erro, setErro] = useState("");
  const [sucesso, setSucesso] = useState("");

  useEffect(() => {
    void carregarCadastros();
  }, []);

  async function carregarCadastros() {
    try {
      setCarregandoCadastros(true);

      const {
        data: { user },
      } = await supabase.auth.getUser();

      if (!user) {
        setRedirecionandoLogin(true);
        router.replace("/login" as any);
        return;
      }

      setValidandoSessao(false);

      const [clientesResult, obrasResult, servicosResult] = await Promise.all([
        supabase
          .from("clientes")
          .select("id, nome, telefone, email, endereco, cidade, estado")
          .eq("user_id", user.id)
          .order("nome", { ascending: true }),
        supabase
          .from("obras")
          .select("id, nome, cliente_id, status")
          .eq("user_id", user.id)
          .order("created_at", { ascending: false }),
        supabase
          .from("servicos")
          .select("id, nome, unidade, valor_unitario, ativo")
          .eq("user_id", user.id)
          .eq("ativo", true)
          .order("nome", { ascending: true }),
      ]);

      if (!clientesResult.error) {
        setClientesCadastrados(
          (clientesResult.data ?? []) as ClienteCadastrado[]
        );
      }

      if (!obrasResult.error) {
        setObrasCadastradas((obrasResult.data ?? []) as ObraCadastrada[]);
      }

      if (!servicosResult.error) {
        setServicosCadastrados(
          (servicosResult.data ?? []) as ServicoCadastrado[]
        );
      }
    } catch (error) {
      console.log("Erro ao carregar cadastros do orçamento:", error);
    } finally {
      setCarregandoCadastros(false);
    }
  }

  function selecionarCliente(cliente: ClienteCadastrado) {
    setClienteSelecionadoId(cliente.id);
    setClienteNome(cliente.nome || "");
    setClienteTelefone(cliente.telefone || "");
    setClienteEmail(cliente.email || "");

    // O cadastro atual de clientes possui endereço consolidado.
    // Mantemos os campos já digitados de CEP/número quando não houver dados
    // suficientes para separar o endereço automaticamente.
    if (cliente.cidade) {
      setClienteCidade(cliente.cidade);
    }

    if (cliente.estado) {
      setClienteUf(cliente.estado.slice(0, 2).toUpperCase());
      setClienteEstado(cliente.estado);
    }
  }

  function selecionarServico(
    itemId: string,
    servico: ServicoCadastrado
  ) {
    const valor = Number(servico.valor_unitario || 0);

    setItens((atuais) =>
      atuais.map((item) =>
        item.id === itemId
          ? {
              ...item,
              servicoId: servico.id,
              descricao: servico.nome,
              valorUnitario: Number.isFinite(valor)
                ? valor.toFixed(2).replace(".", ",")
                : "",
            }
          : item
      )
    );
  }

  const subtotal = useMemo(() => {
    return itens.reduce((acc, item) => {
      const quantidade = normalizarNumero(item.quantidade);
      const valorUnitario = normalizarNumero(item.valorUnitario);

      return acc + quantidade * valorUnitario;
    }, 0);
  }, [itens]);

  const descontoNumero = useMemo(() => {
    const valor = normalizarNumero(desconto);
    return valor < 0 ? 0 : valor;
  }, [desconto]);

  const acrescimoNumero = useMemo(() => {
    const valor = normalizarNumero(acrescimo);
    return valor < 0 ? 0 : valor;
  }, [acrescimo]);

  const total = Math.max(subtotal + acrescimoNumero - descontoNumero, 0);

  function formatarCep(valor: string) {
    const numeros = valor.replace(/\D/g, "").slice(0, 8);

    if (numeros.length <= 5) {
      return numeros;
    }

    return `${numeros.slice(0, 5)}-${numeros.slice(5)}`;
  }

  function limparEnderecoCep() {
    setClienteLogradouro("");
    setClienteBairro("");
    setClienteCidade("");
    setClienteUf("");
    setClienteEstado("");
  }

  async function buscarCep(valor: string) {
    const cep = valor.replace(/\D/g, "");

    if (cep.length !== 8) {
      return;
    }

    try {
      setBuscandoCep(true);
      setErroCep("");

      const resposta = await fetch(`https://viacep.com.br/ws/${cep}/json/`);

      if (!resposta.ok) {
        limparEnderecoCep();
        setErroCep("Não foi possível consultar este CEP.");
        return;
      }

      const dados = await resposta.json();

      if (dados.erro) {
        limparEnderecoCep();
        setErroCep("CEP não encontrado.");
        return;
      }

      setClienteCep(formatarCep(dados.cep || cep));
      setClienteLogradouro(dados.logradouro || "");
      setClienteBairro(dados.bairro || "");
      setClienteCidade(dados.localidade || "");
      setClienteUf(dados.uf || "");
      setClienteEstado(dados.estado || "");
    } catch {
      limparEnderecoCep();
      setErroCep("Não foi possível consultar o CEP. Verifique sua conexão.");
    } finally {
      setBuscandoCep(false);
    }
  }

  function atualizarItem(
    id: string,
    campo: "descricao" | "quantidade" | "valorUnitario",
    valor: string
  ) {
    setItens((atuais) =>
      atuais.map((item) =>
        item.id === id
          ? {
              ...item,
              [campo]: valor,
              ...(campo === "descricao" || campo === "valorUnitario"
                ? { servicoId: null }
                : {}),
            }
          : item
      )
    );
  }

  function adicionarItem() {
    setItens((atuais) => [
      ...atuais,
      {
        id: criarIdLocal(),
        descricao: "",
        quantidade: "1",
        valorUnitario: "",
        servicoId: null,
      },
    ]);
  }

  function removerItem(id: string) {
    setItens((atuais) => {
      if (atuais.length === 1) {
        return atuais;
      }

      return atuais.filter((item) => item.id !== id);
    });
  }

  function validar() {
    if (!clienteNome.trim()) {
      setErro("Informe o nome do cliente.");
      return false;
    }

    if (clienteCep.replace(/\D/g, "").length !== 8) {
      setErro("Informe um CEP válido com 8 dígitos.");
      return false;
    }

    if (!clienteLogradouro && !clienteCidade) {
      setErro("Consulte o CEP antes de salvar o orçamento.");
      return false;
    }

    if (!clienteNumero.trim()) {
      setErro("Informe o número do endereço do cliente.");
      return false;
    }

    if (!titulo.trim()) {
      setErro("Informe um título para o orçamento.");
      return false;
    }

    const validade = Number(validadeDias);

    if (!Number.isInteger(validade) || validade <= 0) {
      setErro("Informe uma validade em dias válida.");
      return false;
    }

    if (itens.length === 0) {
      setErro("Adicione pelo menos um item ao orçamento.");
      return false;
    }

    for (let i = 0; i < itens.length; i += 1) {
      const item = itens[i];
      const quantidade = normalizarNumero(item.quantidade);
      const valorUnitario = normalizarNumero(item.valorUnitario);

      if (!item.descricao.trim()) {
        setErro(`Informe a descrição do item ${i + 1}.`);
        return false;
      }

      if (quantidade <= 0) {
        setErro(`A quantidade do item ${i + 1} precisa ser maior que zero.`);
        return false;
      }

      if (valorUnitario < 0) {
        setErro(`O valor do item ${i + 1} não pode ser negativo.`);
        return false;
      }
    }

    if (subtotal <= 0) {
      setErro(
        "Informe pelo menos um item ou serviço com valor maior que zero."
      );
      return false;
    }

    if (descontoNumero > subtotal + acrescimoNumero) {
      setErro(
        "O desconto não pode ser maior que o valor dos itens somado ao acréscimo."
      );
      return false;
    }

    setErro("");
    return true;
  }

  async function salvarOrcamento() {
    if (!validar()) {
      return;
    }

    let orcamentoCriadoId: string | null = null;

    try {
      setSalvando(true);
      setErro("");
      setSucesso("");

      const {
        data: { user },
        error: userError,
      } = await supabase.auth.getUser();

      if (userError || !user) {
        setErro("Sua sessão expirou. Entre novamente.");
        return;
      }

      const { data: orcamento, error: orcamentoError } = await supabase
        .from("orcamentos")
        .insert({
          user_id: user.id,
          cliente_id: clienteSelecionadoId,
          obra_id: obraSelecionadaId,
          cliente_nome: clienteNome.trim(),
          cliente_telefone: clienteTelefone.trim() || null,
          cliente_cep: clienteCep.replace(/\D/g, "") || null,
          cliente_logradouro: clienteLogradouro.trim() || null,
          cliente_numero: clienteNumero.trim() || null,
          cliente_complemento: clienteComplemento.trim() || null,
          cliente_bairro: clienteBairro.trim() || null,
          cliente_cidade: clienteCidade.trim() || null,
          cliente_uf: clienteUf.trim() || null,
          cliente_estado: clienteEstado.trim() || null,
          cliente_endereco:
            [
              clienteLogradouro.trim(),
              clienteNumero.trim(),
              clienteComplemento.trim(),
              clienteBairro.trim(),
              clienteCidade.trim(),
              clienteUf.trim(),
            ]
              .filter(Boolean)
              .join(", ") || null,
          cliente_email: clienteEmail.trim().toLowerCase() || null,
          titulo: titulo.trim(),
          descricao: descricao.trim() || null,
          observacoes: observacoes.trim() || null,
          validade_dias: Number(validadeDias),
          desconto: descontoNumero,
          acrescimo: acrescimoNumero,
          forma_pagamento: formaPagamento.trim() || null,
          condicoes_pagamento: condicoesPagamento.trim() || null,
          subtotal,
          total,
          status: "rascunho",
        })
        .select("id, numero")
        .single();

      if (orcamentoError || !orcamento) {
        setErro(orcamentoError?.message || "Não foi possível salvar o orçamento.");
        return;
      }

      orcamentoCriadoId = orcamento.id;

      const itensParaSalvar = itens.map((item, index) => {
        const quantidade = normalizarNumero(item.quantidade);
        const valorUnitario = normalizarNumero(item.valorUnitario);

        return {
          orcamento_id: orcamento.id,
          user_id: user.id,
          servico_id: item.servicoId,
          descricao: item.descricao.trim(),
          quantidade,
          valor_unitario: valorUnitario,
          total: quantidade * valorUnitario,
          ordem: index,
        };
      });

      const { error: itensError } = await supabase
        .from("orcamento_itens")
        .insert(itensParaSalvar);

      if (itensError) {
        await supabase.from("orcamentos").delete().eq("id", orcamento.id);

        setErro(
          `Não foi possível salvar os itens do orçamento: ${itensError.message}`
        );
        return;
      }

      setSucesso(
        `Orçamento #${orcamento.numero} salvo com sucesso.`
      );

      setTimeout(() => {
        router.replace("/orcamentos" as any);
      }, 900);
    } catch {
      if (orcamentoCriadoId) {
        await supabase.from("orcamentos").delete().eq("id", orcamentoCriadoId);
      }

      setErro("Ocorreu um erro ao salvar o orçamento. Tente novamente.");
    } finally {
      setSalvando(false);
    }
  }

  if (validandoSessao || redirecionandoLogin) {
    return (
      <SafeAreaView style={styles.safeArea}>
        <View style={styles.authLoading}>
          <ActivityIndicator size="large" color="#7C5CFF" />
        </View>
      </SafeAreaView>
    );
  }

  return (
    <SafeAreaView style={styles.safeArea}>
      <KeyboardAvoidingView
        style={styles.flex}
        behavior={Platform.OS === "ios" ? "padding" : undefined}
      >
        <ScrollView
          contentContainerStyle={styles.scrollContent}
          keyboardShouldPersistTaps="handled"
          showsVerticalScrollIndicator={false}
        >
          <View style={styles.page}>
            <View style={styles.header}>
              <Pressable style={styles.backButton} onPress={() => router.replace("/orcamentos" as any)}>
                <Text style={styles.backButtonText}>←</Text>
              </Pressable>

              <View style={styles.headerTextArea}>
                <Text style={styles.headerEyebrow}>ORVENA</Text>
                <Text style={styles.headerTitle}>Novo orçamento</Text>
                <Text style={styles.headerSubtitle}>
                  Preencha os dados abaixo para criar um novo orçamento.
                </Text>
              </View>
            </View>

            <View
              style={[
                styles.columns,
                isDesktop && styles.columnsDesktop,
              ]}
            >
              <View style={styles.mainColumn}>
                <SectionCard
                  title="Cliente"
                  subtitle="Informações de quem receberá o orçamento."
                >
                  {carregandoCadastros ? (
                    <View style={styles.catalogLoading}>
                      <ActivityIndicator size="small" color="#8F79FF" />
                      <Text style={styles.catalogLoadingText}>
                        Carregando seus cadastros...
                      </Text>
                    </View>
                  ) : clientesCadastrados.length > 0 ? (
                    <View style={styles.catalogBlock}>
                      <Text style={styles.catalogLabel}>
                        USAR CLIENTE CADASTRADO
                      </Text>

                      <ScrollView
                        horizontal
                        showsHorizontalScrollIndicator={false}
                        contentContainerStyle={styles.catalogChips}
                      >
                        {clientesCadastrados.map((cliente) => (
                          <Pressable
                            key={cliente.id}
                            style={[
                              styles.catalogChip,
                              clienteSelecionadoId === cliente.id &&
                                styles.catalogChipActive,
                            ]}
                            onPress={() => selecionarCliente(cliente)}
                          >
                            <Text
                              style={[
                                styles.catalogChipText,
                                clienteSelecionadoId === cliente.id &&
                                  styles.catalogChipTextActive,
                              ]}
                            >
                              {cliente.nome}
                            </Text>
                          </Pressable>
                        ))}

                        {clienteSelecionadoId ? (
                          <Pressable
                            style={styles.catalogChip}
                            onPress={() => setClienteSelecionadoId(null)}
                          >
                            <Text style={styles.catalogChipText}>
                              DIGITAR OUTRO
                            </Text>
                          </Pressable>
                        ) : null}
                      </ScrollView>

                      <Text style={styles.catalogHint}>
                        Ao escolher um cliente, os dados disponíveis são
                        preenchidos automaticamente.
                      </Text>
                    </View>
                  ) : null}

                  <FieldLabel text="Nome do cliente *" />
                  <TextInput
                    value={clienteNome}
                    onChangeText={setClienteNome}
                    placeholder="Ex.: João da Silva"
                    placeholderTextColor="#545A65"
                    style={styles.input}
                  />

                  <View
                    style={[
                      styles.fieldGrid,
                      isDesktop && styles.fieldGridDesktop,
                    ]}
                  >
                    <View style={styles.fieldGridItem}>
                      <FieldLabel text="Telefone" />
                      <TextInput
                        value={clienteTelefone}
                        onChangeText={setClienteTelefone}
                        placeholder="(81) 99999-9999"
                        placeholderTextColor="#545A65"
                        keyboardType="phone-pad"
                        style={styles.input}
                      />
                    </View>

                    <View style={styles.fieldGridItem}>
                      <FieldLabel text="E-mail (opcional)" />
                      <TextInput
                        value={clienteEmail}
                        onChangeText={setClienteEmail}
                        placeholder="cliente@email.com"
                        placeholderTextColor="#545A65"
                        keyboardType="email-address"
                        autoCapitalize="none"
                        autoCorrect={false}
                        style={styles.input}
                      />
                    </View>
                  </View>

                  <FieldLabel text="CEP" />
                  <View style={styles.cepRow}>
                    <TextInput
                      value={clienteCep}
                      onChangeText={(valor) => {
                        const formatado = formatarCep(valor);
                        setClienteCep(formatado);
                        setErroCep("");

                        if (formatado.replace(/\D/g, "").length === 8) {
                          void buscarCep(formatado);
                        } else {
                          limparEnderecoCep();
                        }
                      }}
                      placeholder="00000-000"
                      placeholderTextColor="#545A65"
                      keyboardType="number-pad"
                      maxLength={9}
                      style={[styles.input, styles.cepInput]}
                    />

                    <Pressable
                      style={[
                        styles.cepButton,
                        buscandoCep && styles.cepButtonDisabled,
                      ]}
                      onPress={() => buscarCep(clienteCep)}
                      disabled={buscandoCep}
                    >
                      {buscandoCep ? (
                        <ActivityIndicator size="small" color="#FFFFFF" />
                      ) : (
                        <Text style={styles.cepButtonText}>BUSCAR</Text>
                      )}
                    </Pressable>
                  </View>

                  {erroCep ? (
                    <Text style={styles.cepError}>{erroCep}</Text>
                  ) : null}

                  {(clienteLogradouro ||
                    clienteBairro ||
                    clienteCidade ||
                    clienteUf) ? (
                    <View style={styles.addressResult}>
                      <View style={styles.addressResultHeader}>
                        <Text style={styles.addressResultTitle}>
                          Endereço encontrado
                        </Text>
                        <Text style={styles.addressResultCheck}>✓</Text>
                      </View>

                      <Text style={styles.addressResultText}>
                        {[
                          clienteLogradouro,
                          clienteBairro,
                          clienteCidade,
                          clienteUf,
                        ]
                          .filter(Boolean)
                          .join(" • ")}
                      </Text>

                      {clienteEstado ? (
                        <Text style={styles.addressResultState}>
                          {clienteEstado}
                        </Text>
                      ) : null}
                    </View>
                  ) : null}

                  <View
                    style={[
                      styles.fieldGrid,
                      isDesktop && styles.fieldGridDesktop,
                    ]}
                  >
                    <View style={styles.fieldGridItem}>
                      <FieldLabel text="Número *" />
                      <TextInput
                        value={clienteNumero}
                        onChangeText={setClienteNumero}
                        placeholder="Ex.: 125"
                        placeholderTextColor="#545A65"
                        style={styles.input}
                      />
                    </View>

                    <View style={styles.fieldGridItem}>
                      <FieldLabel text="Complemento (opcional)" />
                      <TextInput
                        value={clienteComplemento}
                        onChangeText={setClienteComplemento}
                        placeholder="Apto, bloco, casa..."
                        placeholderTextColor="#545A65"
                        style={styles.input}
                      />
                    </View>
                  </View>
                </SectionCard>

                <SectionCard
                  title="Orçamento"
                  subtitle="Defina o serviço, validade e informações gerais."
                >
                  {obrasCadastradas.length > 0 ? (
                    <View style={styles.catalogBlock}>
                      <Text style={styles.catalogLabel}>
                        VINCULAR A UMA OBRA
                      </Text>

                      <ScrollView
                        horizontal
                        showsHorizontalScrollIndicator={false}
                        contentContainerStyle={styles.catalogChips}
                      >
                        <Pressable
                          style={[
                            styles.catalogChip,
                            !obraSelecionadaId && styles.catalogChipActive,
                          ]}
                          onPress={() => setObraSelecionadaId(null)}
                        >
                          <Text
                            style={[
                              styles.catalogChipText,
                              !obraSelecionadaId &&
                                styles.catalogChipTextActive,
                            ]}
                          >
                            SEM OBRA
                          </Text>
                        </Pressable>

                        {obrasCadastradas.map((obra) => (
                          <Pressable
                            key={obra.id}
                            style={[
                              styles.catalogChip,
                              obraSelecionadaId === obra.id &&
                                styles.catalogChipActive,
                            ]}
                            onPress={() => {
                              setObraSelecionadaId(obra.id);

                              if (
                                obra.cliente_id &&
                                obra.cliente_id !== clienteSelecionadoId
                              ) {
                                const cliente = clientesCadastrados.find(
                                  (item) => item.id === obra.cliente_id
                                );

                                if (cliente) {
                                  selecionarCliente(cliente);
                                }
                              }
                            }}
                          >
                            <Text
                              style={[
                                styles.catalogChipText,
                                obraSelecionadaId === obra.id &&
                                  styles.catalogChipTextActive,
                              ]}
                            >
                              {obra.nome}
                            </Text>
                          </Pressable>
                        ))}
                      </ScrollView>
                    </View>
                  ) : null}

                  <FieldLabel text="Título do orçamento *" />
                  <TextInput
                    value={titulo}
                    onChangeText={setTitulo}
                    placeholder="Ex.: Reforma de cozinha"
                    placeholderTextColor="#545A65"
                    style={styles.input}
                  />

                  <FieldLabel text="Descrição" />
                  <TextInput
                    value={descricao}
                    onChangeText={setDescricao}
                    placeholder="Descreva brevemente o serviço..."
                    placeholderTextColor="#545A65"
                    multiline
                    textAlignVertical="top"
                    style={[styles.input, styles.textArea]}
                  />

                  <FieldLabel text="Validade (dias)" />
                  <TextInput
                    value={validadeDias}
                    onChangeText={(valor) =>
                      setValidadeDias(valor.replace(/\D/g, ""))
                    }
                    placeholder="15"
                    placeholderTextColor="#545A65"
                    keyboardType="number-pad"
                    style={styles.input}
                  />
                </SectionCard>

                <SectionCard
                  title="Itens"
                  subtitle="Adicione produtos ou serviços ao orçamento."
                >
                  <View style={styles.itemsList}>
                    {itens.map((item, index) => {
                      const itemTotal =
                        normalizarNumero(item.quantidade) *
                        normalizarNumero(item.valorUnitario);

                      return (
                        <View key={item.id} style={styles.itemCard}>
                          <View style={styles.itemHeader}>
                            <Text style={styles.itemTitle}>
                              Item {index + 1}
                            </Text>

                            {itens.length > 1 ? (
                              <Pressable
                                style={styles.removeButton}
                                onPress={() => removerItem(item.id)}
                              >
                                <Text style={styles.removeButtonText}>
                                  REMOVER
                                </Text>
                              </Pressable>
                            ) : null}
                          </View>

                          {servicosCadastrados.length > 0 ? (
                            <View style={styles.itemServicePicker}>
                              <Text style={styles.catalogLabel}>
                                USAR SERVIÇO CADASTRADO
                              </Text>

                              <ScrollView
                                horizontal
                                showsHorizontalScrollIndicator={false}
                                contentContainerStyle={styles.catalogChips}
                              >
                                {servicosCadastrados.map((servico) => (
                                  <Pressable
                                    key={servico.id}
                                    style={[
                                      styles.catalogChip,
                                      item.servicoId === servico.id &&
                                        styles.catalogChipActive,
                                    ]}
                                    onPress={() =>
                                      selecionarServico(item.id, servico)
                                    }
                                  >
                                    <Text
                                      style={[
                                        styles.catalogChipText,
                                        item.servicoId === servico.id &&
                                          styles.catalogChipTextActive,
                                      ]}
                                    >
                                      {servico.nome} • {formatarMoeda(
                                        Number(servico.valor_unitario || 0)
                                      )}
                                    </Text>
                                  </Pressable>
                                ))}
                              </ScrollView>
                            </View>
                          ) : null}

                          <FieldLabel text="Descrição *" />
                          <TextInput
                            value={item.descricao}
                            onChangeText={(valor) =>
                              atualizarItem(item.id, "descricao", valor)
                            }
                            placeholder="Ex.: Instalação elétrica"
                            placeholderTextColor="#545A65"
                            style={styles.input}
                          />

                          <View
                            style={[
                              styles.fieldGrid,
                              isDesktop && styles.fieldGridDesktop,
                            ]}
                          >
                            <View style={styles.fieldGridItem}>
                              <FieldLabel text="Quantidade *" />
                              <TextInput
                                value={item.quantidade}
                                onChangeText={(valor) =>
                                  atualizarItem(item.id, "quantidade", valor)
                                }
                                placeholder="1"
                                placeholderTextColor="#545A65"
                                keyboardType="decimal-pad"
                                style={styles.input}
                              />
                            </View>

                            <View style={styles.fieldGridItem}>
                              <FieldLabel text="Valor unitário (R$) *" />
                              <TextInput
                                value={item.valorUnitario}
                                onChangeText={(valor) =>
                                  atualizarItem(item.id, "valorUnitario", valor)
                                }
                                placeholder="0,00"
                                placeholderTextColor="#545A65"
                                keyboardType="decimal-pad"
                                style={styles.input}
                              />
                            </View>
                          </View>

                          <View style={styles.itemTotalRow}>
                            <Text style={styles.itemTotalLabel}>
                              Total do item
                            </Text>
                            <Text style={styles.itemTotalValue}>
                              {formatarMoeda(itemTotal)}
                            </Text>
                          </View>
                        </View>
                      );
                    })}
                  </View>

                  <Pressable
                    style={styles.addItemButton}
                    onPress={adicionarItem}
                  >
                    <Text style={styles.addItemButtonText}>
                      + ADICIONAR ITEM
                    </Text>
                  </Pressable>
                </SectionCard>

                <SectionCard
                  title="Valores e pagamento"
                  subtitle="Ajuste o total e informe como o cliente fará o pagamento."
                >
                  <View
                    style={[
                      styles.fieldGrid,
                      isDesktop && styles.fieldGridDesktop,
                    ]}
                  >
                    <View style={styles.fieldGridItem}>
                      <FieldLabel text="Acréscimo / taxa extra (R$)" />
                      <TextInput
                        value={acrescimo}
                        onChangeText={setAcrescimo}
                        placeholder="0,00"
                        placeholderTextColor="#545A65"
                        keyboardType="decimal-pad"
                        style={styles.input}
                      />
                    </View>

                    <View style={styles.fieldGridItem}>
                      <FieldLabel text="Desconto (R$)" />
                      <TextInput
                        value={desconto}
                        onChangeText={setDesconto}
                        placeholder="0,00"
                        placeholderTextColor="#545A65"
                        keyboardType="decimal-pad"
                        style={styles.input}
                      />
                    </View>
                  </View>

                  <FieldLabel text="Forma de pagamento" />
                  <TextInput
                    value={formaPagamento}
                    onChangeText={setFormaPagamento}
                    placeholder="Ex.: Pix, dinheiro, cartão..."
                    placeholderTextColor="#545A65"
                    style={styles.input}
                  />

                  <FieldLabel text="Condições de pagamento" />
                  <TextInput
                    value={condicoesPagamento}
                    onChangeText={setCondicoesPagamento}
                    placeholder="Ex.: 100% no Pix, 2x no cartão, pagamento na entrega..."
                    placeholderTextColor="#545A65"
                    multiline
                    textAlignVertical="top"
                    style={[styles.input, styles.paymentConditionsArea]}
                  />
                </SectionCard>

                <SectionCard
                  title="Observações"
                  subtitle="Informações extras que devem constar no orçamento."
                >
                  <TextInput
                    value={observacoes}
                    onChangeText={setObservacoes}
                    placeholder="Ex.: Materiais inclusos, condições, prazo de execução..."
                    placeholderTextColor="#545A65"
                    multiline
                    textAlignVertical="top"
                    style={[styles.input, styles.textAreaLarge]}
                  />
                </SectionCard>
              </View>

              <View
                style={[
                  styles.sideColumn,
                  isDesktop && styles.sideColumnDesktop,
                ]}
              >
                <View style={styles.summaryCard}>
                  <Text style={styles.summaryEyebrow}>RESUMO</Text>
                  <Text style={styles.summaryTitle}>Valores</Text>

                  <View style={styles.summaryDivider} />

                  <SummaryRow
                    label="Valor dos itens/serviços"
                    value={formatarMoeda(subtotal)}
                  />

                  <View style={styles.summaryDivider} />

                  {acrescimoNumero > 0 ? (
                    <SummaryRow
                      label="Acréscimo"
                      value={`+ ${formatarMoeda(acrescimoNumero)}`}
                    />
                  ) : null}

                  {descontoNumero > 0 ? (
                    <SummaryRow
                      label="Desconto"
                      value={`- ${formatarMoeda(descontoNumero)}`}
                    />
                  ) : null}

                  <View style={styles.summaryDivider} />

                  <View style={styles.totalRow}>
                    <Text style={styles.totalLabel}>Total do orçamento</Text>
                    <Text style={styles.totalValue}>
                      {formatarMoeda(total)}
                    </Text>
                  </View>

                  <Text style={styles.totalFormula}>
                    Itens/serviços + acréscimos - descontos
                  </Text>

                  {(formaPagamento.trim() || condicoesPagamento.trim()) ? (
                    <View style={styles.paymentSummaryBox}>
                      <Text style={styles.paymentSummaryTitle}>
                        PAGAMENTO
                      </Text>

                      {formaPagamento.trim() ? (
                        <View style={styles.paymentSummaryRow}>
                          <Text style={styles.paymentSummaryLabel}>
                            Forma
                          </Text>
                          <Text style={styles.paymentSummaryValue}>
                            {formaPagamento.trim()}
                          </Text>
                        </View>
                      ) : null}

                      {condicoesPagamento.trim() ? (
                        <View style={styles.paymentConditionsSummary}>
                          <Text style={styles.paymentSummaryLabel}>
                            Condições
                          </Text>
                          <Text style={styles.paymentConditionsText}>
                            {condicoesPagamento.trim()}
                          </Text>
                        </View>
                      ) : null}
                    </View>
                  ) : null}

                  <View style={styles.summaryInfo}>
                    <Text style={styles.summaryInfoTitle}>
                      Plano FREE
                    </Text>
                    <Text style={styles.summaryInfoText}>
                      O PDF deste orçamento usará a identidade visual padrão do
                      Orvena.
                    </Text>
                  </View>

                  {erro ? (
                    <View style={styles.errorBox}>
                      <Text style={styles.errorText}>{erro}</Text>
                    </View>
                  ) : null}

                  {sucesso ? (
                    <View style={styles.successBox}>
                      <Text style={styles.successText}>{sucesso}</Text>
                    </View>
                  ) : null}

                  <Pressable
                    style={[
                      styles.saveButton,
                      salvando && styles.saveButtonDisabled,
                    ]}
                    onPress={salvarOrcamento}
                    disabled={salvando}
                  >
                    {salvando ? (
                      <ActivityIndicator color="#FFFFFF" />
                    ) : (
                      <Text style={styles.saveButtonText}>
                        SALVAR ORÇAMENTO
                      </Text>
                    )}
                  </Pressable>

                  <Pressable
                    style={styles.cancelButton}
                    onPress={() => router.replace("/orcamentos" as any)}
                    disabled={salvando}
                  >
                    <Text style={styles.cancelButtonText}>CANCELAR</Text>
                  </Pressable>
                </View>
              </View>
            </View>
          </View>
        </ScrollView>
      </KeyboardAvoidingView>
    </SafeAreaView>
  );
}

function SectionCard({
  title,
  subtitle,
  children,
}: {
  title: string;
  subtitle: string;
  children: React.ReactNode;
}) {
  return (
    <View style={styles.sectionCard}>
      <View style={styles.sectionHeader}>
        <Text style={styles.sectionTitle}>{title}</Text>
        <Text style={styles.sectionSubtitle}>{subtitle}</Text>
      </View>

      {children}
    </View>
  );
}

function FieldLabel({ text }: { text: string }) {
  return <Text style={styles.fieldLabel}>{text}</Text>;
}

function SummaryRow({
  label,
  value,
}: {
  label: string;
  value: string;
}) {
  return (
    <View style={styles.summaryRow}>
      <Text style={styles.summaryRowLabel}>{label}</Text>
      <Text style={styles.summaryRowValue}>{value}</Text>
    </View>
  );
}
